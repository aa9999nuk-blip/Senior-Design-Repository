// ppg_ble_lightsleep.ino // // Power-saving rewrite using: // - BLE kept alive while connected // - light sleep for idle connected periods // - deep sleep only when disconnected and outside the reconnect window // // Timekeeping: // RTC_DATA_ATTR keeps a wall-clock accumulator across deep sleep resets. // Light sleep adds actual slept time to the accumulator. #include #include #include #include #include #include #include "SparkFun_Bio_Sensor_Hub_Library.h" #include "esp_sleep.h" #include "esp_timer.h" // ─── Pin / address config ───────────────────────────────────────────────────── const int RESET_PIN = 17; const int MFIO_PIN = 18; const int I2C_ADDRESS = 0x55; // ─── Timing ─────────────────────────────────────────────────────────────────── // Deep-sleep interval when idle and disconnected. static const uint64_t WAKE_INTERVAL_SEC = 60ULL; // How long to keep BLE advertising alive after boot or disconnect. static const uint32_t BLE_WINDOW_MS = 15000UL; // Short nap while connected and idle. // Keep this short so BLE connection events are not missed. static const uint32_t CONNECTED_LIGHT_SLEEP_MS = 20UL; // Duration of one PPG measurement session. static const uint32_t MEASURE_DURATION_MS = 30000UL; // Auto-measurement interval, tracked as wall-clock time. static const uint32_t AUTO_MEASURE_INTERVAL_SEC = 15UL * 60UL; // ─── RTC memory — survives deep sleep ───────────────────────────────────────── // Accumulated elapsed time in milliseconds across deep sleep + light sleep. RTC_DATA_ATTR static uint64_t rtcElapsedMs = 0; // Wall-clock timestamp of the last completed measurement session. RTC_DATA_ATTR static uint64_t rtcLastMeasureMs = 0; // Used to detect cold boot vs deep-sleep wake. RTC_DATA_ATTR static bool rtcFirstBoot = true; // ─── BLE UUIDs ──────────────────────────────────────────────────────────────── #define SERVICE_UUID "12345678-1234-1234-1234-123456789012" #define CHARACTERISTIC_UUID "87654321-4321-4321-4321-210987654321" #define COMMAND_CHARACTERISTIC_UUID "11223344-5566-7788-9900-aabbccddeeff" BLECharacteristic *pDataChar = nullptr; BLECharacteristic *pCommandChar = nullptr; volatile bool deviceConnected = false; volatile bool measureRequested = false; SparkFun_Bio_Sensor_Hub bioHub(RESET_PIN, MFIO_PIN, I2C_ADDRESS); // Tracks the current reconnect / advertising window. uint32_t bleWindowStartMs = 0; // ─── Helpers ───────────────────────────────────────────────────────────────── bool isAutoMeasureDue() { uint64_t sinceLast = (rtcElapsedMs >= rtcLastMeasureMs) ? (rtcElapsedMs - rtcLastMeasureMs) : 0ULL; return sinceLast >= (uint64_t)AUTO_MEASURE_INTERVAL_SEC * 1000ULL; } bool bleWindowOpen() { return (millis() - bleWindowStartMs) < BLE_WINDOW_MS; } void lightSleepFor(uint32_t ms) { uint64_t before = esp_timer_get_time(); // microseconds since boot esp_sleep_enable_timer_wakeup((uint64_t)ms * 1000ULL); esp_light_sleep_start(); // returns on wake uint64_t after = esp_timer_get_time(); if (after > before) { rtcElapsedMs += (after - before) / 1000ULL; } else { rtcElapsedMs += ms; } } void goToDeepSleep(uint64_t seconds) { rtcElapsedMs += seconds * 1000ULL; uint64_t sinceLast = (rtcElapsedMs >= rtcLastMeasureMs) ? (rtcElapsedMs - rtcLastMeasureMs) : 0ULL; uint64_t secUntilNext = 0; uint64_t intervalMs = (uint64_t)AUTO_MEASURE_INTERVAL_SEC * 1000ULL; if (sinceLast < intervalMs) { secUntilNext = (intervalMs - sinceLast) / 1000ULL; } Serial.printf("[SLEEP] Accumulated: %llu ms | Next auto-measure in: ~%llu s\n", (unsigned long long)rtcElapsedMs, (unsigned long long)secUntilNext); Serial.println("[SLEEP] Shutting down BLE and entering deep sleep..."); Serial.flush(); BLEDevice::deinit(true); esp_sleep_enable_timer_wakeup(seconds * 1000000ULL); esp_deep_sleep_start(); // does not return } // ─── BLE server callbacks ───────────────────────────────────────────────────── class ServerCallbacks : public BLEServerCallbacks { void onConnect(BLEServer *) override { deviceConnected = true; Serial.println("[BLE] Client connected"); } void onDisconnect(BLEServer *) override { deviceConnected = false; Serial.println("[BLE] Client disconnected"); // Reopen the reconnect window after a disconnect. bleWindowStartMs = millis(); BLEDevice::startAdvertising(); } }; // ─── Command characteristic callbacks ───────────────────────────────────────── class CommandCallbacks : public BLECharacteristicCallbacks { void onWrite(BLECharacteristic *pChar) override { std::string val = pChar->getValue(); if (val == "START") { Serial.println("[CMD] START received — measurement requested"); measureRequested = true; } else { Serial.printf("[CMD] Unknown command: %s\n", val.c_str()); } } }; // ─── BLE initialization ─────────────────────────────────────────────────────── void initBLE() { BLEDevice::init("QT_Py_ESP32S3"); BLEDevice::setMTU(512); BLEServer *pServer = BLEDevice::createServer(); pServer->setCallbacks(new ServerCallbacks()); BLEService *pService = pServer->createService(SERVICE_UUID); // Data characteristic — notifies app with "irLed,redLed" strings. pDataChar = pService->createCharacteristic( CHARACTERISTIC_UUID, BLECharacteristic::PROPERTY_READ | BLECharacteristic::PROPERTY_NOTIFY ); pDataChar->addDescriptor(new BLE2902()); pDataChar->setValue("0,0"); // Command characteristic — receives "START" writes from the app. pCommandChar = pService->createCharacteristic( COMMAND_CHARACTERISTIC_UUID, BLECharacteristic::PROPERTY_WRITE ); pCommandChar->setCallbacks(new CommandCallbacks()); pService->start(); BLEAdvertising *pAdv = BLEDevice::getAdvertising(); pAdv->addServiceUUID(SERVICE_UUID); pAdv->setScanResponse(false); pAdv->setMinPreferred(0x0); BLEDevice::startAdvertising(); Serial.println("[BLE] Stack up, advertising..."); } // ─── Sensor initialization ──────────────────────────────────────────────────── bool initSensor() { Wire1.begin(); Wire1.setClock(400000); int result = bioHub.begin(Wire1, RESET_PIN, MFIO_PIN); if (result != 0) { Serial.printf("[SENSOR] Init failed (err %d)\n", result); return false; } bioHub.configSensor(); bioHub.setSampleRate(100); bioHub.setPulseWidth(69); Serial.println("[SENSOR] Init OK"); return true; } // ─── Single sample read + BLE notify ───────────────────────────────────────── void readAndNotify() { bioData body = bioHub.readSensor(); if (deviceConnected) { String msg = String(body.irLed) + "," + String(body.redLed); pDataChar->setValue(msg.c_str()); pDataChar->notify(); Serial.printf("[DATA] IR: %lu | Red: %lu\n", (unsigned long)body.irLed, (unsigned long)body.redLed); } } // ─── 30-second PPG measurement session ─────────────────────────────────────── void runMeasurementSession() { Serial.println("[MEASURE] Starting 30-sec PPG session"); measureRequested = false; if (!initSensor()) { Serial.println("[MEASURE] Skipping — sensor unavailable"); return; } unsigned long sessionStart = millis(); while (millis() - sessionStart < MEASURE_DURATION_MS) { readAndNotify(); if (!deviceConnected) { Serial.println("[MEASURE] Client disconnected — ending early"); break; } delay(10); // ~100 Hz, matching setSampleRate(100) } Serial.println("[MEASURE] Session complete"); } // ─── Setup — runs on every boot / deep-sleep wake ──────────────────────────── void setup() { Serial.begin(115200); delay(100); esp_sleep_wakeup_cause_t wakeReason = esp_sleep_get_wakeup_cause(); if (rtcFirstBoot) { rtcFirstBoot = false; // Force the first auto-measure to be due on cold start. rtcElapsedMs = (uint64_t)AUTO_MEASURE_INTERVAL_SEC * 1000ULL; rtcLastMeasureMs = 0; Serial.println("[BOOT] Cold start — first measurement will fire immediately"); } else if (wakeReason == ESP_SLEEP_WAKEUP_TIMER) { Serial.printf("[WAKE] Timer wake — accumulated time: %llu ms\n", (unsigned long long)rtcElapsedMs); } else { Serial.printf("[WAKE] Wake cause: %d\n", wakeReason); } bleWindowStartMs = millis(); initBLE(); } // ─── Main loop ─────────────────────────────────────────────────────────────── void loop() { const bool due = isAutoMeasureDue(); const bool windowOpen = bleWindowOpen(); // Manual START always wins. if (measureRequested) { runMeasurementSession(); rtcLastMeasureMs = rtcElapsedMs; if (!deviceConnected) { goToDeepSleep(WAKE_INTERVAL_SEC); } return; } // Auto-measure: run immediately if connected. // If disconnected, wait during the reconnect window first. if (due) { if (deviceConnected) { Serial.println("[WAKE] Auto-measure due and client connected"); runMeasurementSession(); rtcLastMeasureMs = rtcElapsedMs; return; } if (!windowOpen) { Serial.println("[WAKE] Auto-measure due, no client connected — running anyway"); runMeasurementSession(); rtcLastMeasureMs = rtcElapsedMs; goToDeepSleep(WAKE_INTERVAL_SEC); } else { lightSleepFor(50); } return; } // Connected idle: keep BLE alive, but let the CPU nap briefly. if (deviceConnected) { lightSleepFor(CONNECTED_LIGHT_SLEEP_MS); return; } // Not connected: keep advertising during the reconnect window. if (windowOpen) { lightSleepFor(50); return; } // No connection and no work to do — sleep deeply until the next cycle. goToDeepSleep(WAKE_INTERVAL_SEC); }
